cmake_minimum_required(VERSION 3.16)
project(scenario_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
file(GLOB SOURCES CONFIGURE_DEPENDS ../../components/scenario_engine/*.cpp)
add_library(cjson_host ../../components/scenario_engine/vendor/cjson/cJSON.c)
target_compile_definitions(cjson_host PRIVATE CJSON_NESTING_LIMIT=32)
target_include_directories(cjson_host PUBLIC ../../components/scenario_engine/vendor/cjson)
add_library(scenario_host ../../components/core/device_id.cpp ../../components/service/scenario_event_adapter.cpp ${SOURCES})
target_include_directories(scenario_host PUBLIC ../../components/core/include ../../components/scenario_engine/include)
target_link_libraries(scenario_host PUBLIC cjson_host)
target_compile_definitions(scenario_host PUBLIC ZIGBEERUBI_PROTOTYPE=1)
target_include_directories(scenario_host PUBLIC ../../components/service/include)
if(MSVC)
  target_compile_options(cjson_host PRIVATE /utf-8)
  target_compile_definitions(cjson_host PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(scenario_host PUBLIC /utf-8)
  target_compile_options(scenario_host PRIVATE /W4)
  target_compile_definitions(scenario_host PUBLIC _CRT_SECURE_NO_WARNINGS)
else()
  # Кадр стека больше 8 КБ опасен для задач ESP32-C6 (main task 3,5 КБ, сервисная 24 КБ) — ловим это на host-сборке.
  target_compile_options(scenario_host PRIVATE -Wall -Wextra -Werror -Wframe-larger-than=8192)
endif()
file(GLOB TESTS CONFIGURE_DEPENDS test_*.cpp)
foreach(TEST ${TESTS})
 get_filename_component(NAME ${TEST} NAME_WE)
 add_executable(${NAME} ${TEST})
 target_link_libraries(${NAME} scenario_host)
 add_test(NAME ${NAME} COMMAND ${NAME})
endforeach()

add_test(NAME review_manual_override COMMAND test_target_runtime manual)
add_test(NAME review_stale COMMAND test_target_runtime stale)
add_test(NAME review_mixed_report_order COMMAND test_target_runtime order)
