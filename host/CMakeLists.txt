cmake_minimum_required(VERSION 3.16)
project(zigbee_hub_simulator C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(../test/scenarios scenario-tests)
add_executable(zigbee_hub_sim simulator_main.cpp simulator_runtime.cpp file_store.cpp)
target_include_directories(zigbee_hub_sim PRIVATE include)
target_link_libraries(zigbee_hub_sim scenario_host)
if(MSVC)
  add_compile_options(/utf-8)
  target_compile_options(zigbee_hub_sim PRIVATE /utf-8 /W4)
  target_compile_definitions(zigbee_hub_sim PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(zigbee_hub_sim PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(test_file_store ../test/host_sim/test_file_store.cpp file_store.cpp)
target_include_directories(test_file_store PRIVATE include)
target_link_libraries(test_file_store PRIVATE scenario_host)
if(MSVC)
  target_compile_options(test_file_store PRIVATE /utf-8)
  target_compile_definitions(test_file_store PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME test_file_store COMMAND test_file_store)
